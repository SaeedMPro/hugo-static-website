{{- /* Gregorian time → Jalali (Solar Hijri) date as dict "y" "m" "d" (integers). */ -}} {{- $gy := .Year -}} {{- $gm := int (.Format "1") -}} {{- $gd := .Day -}} {{- $before := slice 0 31 59 90 120 151 181 212 243 273 304 334 -}} {{- $gy2 := cond (gt $gm 2) (add $gy 1) $gy -}} {{- $days := add 355666 (mul 365 $gy) (div (add $gy2 3) 4) (mul -1 (div (add $gy2 99) 100)) (div (add $gy2 399) 400) $gd (index $before (sub $gm 1)) -}} {{- $jy := add -1595 (mul 33 (div $days 12053)) -}} {{- $days = mod $days 12053 -}} {{- $jy = add $jy (mul 4 (div $days 1461)) -}} {{- $days = mod $days 1461 -}} {{- if gt $days 365 -}} {{- $jy = add $jy (div (sub $days 1) 365) -}} {{- $days = mod (sub $days 1) 365 -}} {{- end -}} {{- $jm := 0 -}} {{- $jd := 0 -}} {{- if lt $days 186 -}} {{- $jm = add 1 (div $days 31) -}} {{- $jd = add 1 (mod $days 31) -}} {{- else -}} {{- $jm = add 7 (div (sub $days 186) 30) -}} {{- $jd = add 1 (mod (sub $days 186) 30) -}} {{- end -}} {{- $res := dict "y" $jy "m" $jm "d" $jd -}} {{- return $res -}}
